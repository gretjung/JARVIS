    # ========================================================
    # SEND
    # ========================================================

    def send_message(self):

        if self.is_processing:
            return

        text = self.entry.get().strip()

        if not text:
            return

        self.entry.delete(
            0,
            "end"
        )

        self.add_chat(
            "USER",
            text
        )

        self.set_status(
            "PROCESSING...",
            "#FFD166"
        )

        self.is_processing = True

        self.send_button.configure(
            state="disabled"
        )

        thread = threading.Thread(
            target=self.process_message,
            args=(text,),
            daemon=True
        )

        thread.start()

    # ========================================================
    # PROCESS
    # ========================================================

    def process_message(
        self,
        text
    ):

        try:

            answer = self.agent.handle(
                text
            )

        except Exception as e:

            answer = (
                "à¹€à¸à¸´à¸”à¸‚à¹‰à¸­à¸œà¸´à¸”à¸žà¸¥à¸²à¸”à¸„à¸£à¸±à¸š\n\n"
                f"{e}"
            )

        # ----------------------------------------------------
        # GUI update must run on main thread
        # ----------------------------------------------------

        self.after(
            0,
            lambda: self.finish(
                answer
            )
        )

    # ========================================================
    # FINISH
    # ========================================================

    def finish(
        self,
        answer
    ):

        # ----------------------------------------------------
        # Absolute protection against bool / None
        # ----------------------------------------------------

        if answer is None:

            answer = ""

        elif not isinstance(
            answer,
            str
        ):

            answer = str(
                answer
            )

        # ----------------------------------------------------
        # Chat
        # ----------------------------------------------------

        self.add_chat(
            "JARVIS",
            answer
        )

        # ----------------------------------------------------
        # Image detection
        # ----------------------------------------------------

        image_path = self.find_image_path(
            answer
        )

        if image_path:

            self.show_image(
                image_path
            )

        # ----------------------------------------------------
        # Status
        # ----------------------------------------------------

        self.set_status(
            "SYSTEM ONLINE",
            "#48FF9B"
        )

        self.is_processing = False

        self.send_button.configure(
            state="normal"
        )

        self.entry.focus_set()

    # ========================================================
    # FIND IMAGE PATH
    # ========================================================

    def find_image_path(
        self,
        text
    ):

        # ----------------------------------------------------
        # à¸›à¹‰à¸­à¸‡à¸à¸±à¸™ None
        # ----------------------------------------------------

        if text is None:

            return None

        # ----------------------------------------------------
        # à¸›à¹‰à¸­à¸‡à¸à¸±à¸™ bool / int / object
        # ----------------------------------------------------

        if not isinstance(
            text,
            str
        ):

            text = str(
                text
            )

        # ====================================================
        # IMAGE_PATH
        # ====================================================

        pattern = (
            r"IMAGE_PATH:\s*(.+)"
        )

        match = re.search(
            pattern,
            text,
            re.IGNORECASE
        )

        if match:

            path = (
                match.group(1)
                .strip()
            )

            path = path.strip(
                '"'
            )

            path = path.strip(
                "'"
            )

            # -----------------------------------------------
            # à¸–à¹‰à¸²à¸¡à¸µà¸‚à¹‰à¸­à¸„à¸§à¸²à¸¡à¸•à¹ˆà¸­à¸—à¹‰à¸²à¸¢ path
            # -----------------------------------------------

            if "\n" in path:

                path = path.split(
                    "\n",
                    1
                )[0].strip()

            if os.path.exists(
                path
            ):

                return path

        # ====================================================
        # PNG
        # ====================================================

        png_pattern = (
            r"([A-Za-z]:[\\/]"
            r"[^<>:\"|?*\r\n]+"
            r"\.png)"
        )

        matches = re.findall(
            png_pattern,
            text,
            re.IGNORECASE
        )

        for path in matches:

            path = path.strip()

            path = path.strip(
                '"'
            )

            path = path.strip(
                "'"
            )

            if os.path.exists(
                path
            ):

                return path

        # ====================================================
        # JPG / JPEG / WEBP
        # ====================================================

        image_pattern = (
            r"([A-Za-z]:[\\/]"
            r"[^<>:\"|?*\r\n]+"
            r"\.(?:png|jpg|jpeg|webp))"
        )

        matches = re.findall(
            image_pattern,
            text,
            re.IGNORECASE
        )

        for path in matches:

            path = path.strip()

            path = path.strip(
                '"'
            )

            path = path.strip(
                "'"
            )

            if os.path.exists(
                path
            ):

                return path

        return None

    # ========================================================
    # SHOW IMAGE
    # ========================================================

    def show_image(
        self,
        path
    ):

        try:

            if not path:
                return

            if not os.path.exists(
                path
            ):

                return

            image = Image.open(
                path
            )

            image = image.convert(
                "RGB"
            )

            # ------------------------------------------------
            # Calculate display size
            # ------------------------------------------------

            frame_width = (
                self.image_display_frame
                .winfo_width()
            )

            frame_height = (
                self.image_display_frame
                .winfo_height()
            )

            if frame_width < 100:
                frame_width = 600

            if frame_height < 100:
                frame_height = 600

            max_width = (
                frame_width - 40
            )

            max_height = (
                frame_height - 40
            )

            image.thumbnail(
                (
                    max_width,
                    max_height
                ),
                Image.Resampling.LANCZOS
            )

            ctk_image = ctk.CTkImage(
                light_image=image,
                dark_image=image,
                size=image.size
            )

            self.current_image = (
                ctk_image
            )

            self.current_image_path = (
                path
            )

            self.image_label.configure(
                image=ctk_image,
                text=""
            )

            self.image_label.image = (
                ctk_image
            )

            self.image_title.configure(
                text=(
                    "  IMAGE ENGINE  â€¢  "
                    "IMAGE READY"
                )
            )

        except Exception as e:

            print(
                "[GUI] Image display error:"
            )

            print(
                repr(e)
            )

            self.image_label.configure(
                image=None,
                text=(
                    "IMAGE ERROR\n\n"
                    f"{e}"
                )
            )

    # ========================================================
    # COPY IMAGE
    # ========================================================

    def copy_image(self):

        path = (
            self.current_image_path
        )

        if not path:

            self.set_status(
                "NO IMAGE",
                "#FF6B6B"
            )

            return

        if not os.path.exists(
            path
        ):

            self.set_status(
                "IMAGE NOT FOUND",
                "#FF6B6B"
            )

            return

        if win32clipboard is None:

            self.set_status(
                "PYWIN32 REQUIRED",
                "#FF6B6B"
            )

            return

        try:

            image = Image.open(
                path
            ).convert(
                "RGB"
            )

            output = io.BytesIO()

            image.save(
                output,
                "BMP"
            )

            data = (
                output.getvalue()[14:]
            )

            output.close()

            win32clipboard.OpenClipboard()

            try:

                win32clipboard.EmptyClipboard()

                win32clipboard.SetClipboardData(
                    win32clipboard.CF_DIB,
                    data
                )

            finally:

                win32clipboard.CloseClipboard()

            self.set_status(
                "IMAGE COPIED",
                "#48FF9B"
            )

        except Exception as e:

            print(
                "[GUI] Clipboard error:"
            )

            print(
                repr(e)
            )

            self.set_status(
                "COPY FAILED",
                "#FF6B6B"
            )

    # ========================================================
    # OPEN IMAGE
    # ========================================================

    def open_image(self):

        path = (
            self.current_image_path
        )

        if not path:

            self.set_status(
                "NO IMAGE",
                "#FF6B6B"
            )

            return

        if not os.path.exists(
            path
        ):

            self.set_status(
                "IMAGE NOT FOUND",
                "#FF6B6B"
            )

            return

        try:

            os.startfile(
                path
            )

            self.set_status(
                "IMAGE OPENED",
                "#48FF9B"
            )

        except Exception:

            try:

                subprocess.Popen(
                    [
                        "explorer.exe",
                        "/select,",
                        path
                    ]
                )

                self.set_status(
                    "IMAGE OPENED",
                    "#48FF9B"
                )

            except Exception as e:

                print(
                    "[GUI] "
                    f"Open image error: {e}"
                )

                self.set_status(
                    "OPEN FAILED",
                    "#FF6B6B"
                )
